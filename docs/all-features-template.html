// スライド数を数えるサンプル
function countSlides(html) {
  const matches = html.match(/<div class="slide">/g)
  return matches ? matches.length : 0
}

const total = countSlides(document.body.innerHTML)
console.log(`スライド数: ${total}`)

コードブロック② デフォルト・ミニマル

def fibonacci(n):
    a, b = 0, 1
    for _ in range(n):
        yield a
        a, b = b, a + b

print(list(fibonacci(10)))

リスト4: 透明（slide-code-block-transparent）/ プレーンテキスト

長いコードは最大高さ（max-height: 200px）を指定するとスクロール表示になります。 

interface Slide {
  id: string
  content: string
  pageNumber: number
}

function renumber(slides: Slide[]): Slide[] {
  return slides.map((slide, index) => ({
    ...slide,
    pageNumber: index + 1,
  }))
}

export default renumber
